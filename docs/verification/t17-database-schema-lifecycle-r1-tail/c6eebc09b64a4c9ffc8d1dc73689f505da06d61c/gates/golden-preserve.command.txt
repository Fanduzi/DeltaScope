set -euo pipefail
e=/tmp/t17r1-tail-evidence/gates
src=/tmp/ddl-golden/T17
backup=$(mktemp -d /tmp/ddl-golden-T17-tail-preserved-XXXXXXXX)
printf '%s\n' "$backup/T17" > "$e/golden-preserved-path.txt"
find "$src" -type f -print0 | xargs -0 shasum -a 256 | sed 's#  /tmp/ddl-golden/T17/#  #' | LC_ALL=C sort > "$e/golden-preserve-before.sha256"
mv "$src" "$backup/T17"
find "$backup/T17" -type f -print0 | xargs -0 shasum -a 256 | sed "s#  $backup/T17/#  #" | LC_ALL=C sort > "$e/golden-preserve-after.sha256"
cmp "$e/golden-preserve-before.sha256" "$e/golden-preserve-after.sha256"
printf 'preserved %s with matching before/after file hashes\n' "$backup/T17"
